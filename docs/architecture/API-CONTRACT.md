# GIZER NBA — API contract (v1)

Base path `/api/v1`. JSON only. The web app calls the API same-origin (Next.js rewrites `/api/v1/*`
to the API container). OpenAPI is generated by FastAPI at `/api/v1/openapi.json` (owner only in production).

## Conventions

* Auth: HttpOnly session cookie `gizer_session` set by `POST /auth/login`. Every endpoint except
  `/health`, `/health/ready`, `/auth/login` requires it → `401 {"error":{"code":"unauthenticated","message":…}}`.
* Mutations (`POST`) must send `X-CSRF-Token: <csrf>` (value from `/auth/login` or `/auth/me`) → else `403`.
* Errors always: `{"error": {"code": string, "message": string}}`.
* Timestamps are ISO-8601 UTC. Dates are `YYYY-MM-DD` in America/New_York. Display times in ET.
* Probabilities are floats 0–1. `evidence_score` is an integer 0–100 measuring **evidence quality**, never a probability.
* `null` means "not available" — the UI must show an explicit unavailable state, never a placeholder number.
* Classifications: `high_confidence | standard | watchlist | insufficient_evidence | rejected | invalidated`.
  "Qualified" = `high_confidence` or `standard`.

## Shared shapes

```ts
type TeamRef = { id: number; abbr: string; name: string; logo_url: string | null };

type Selection = {
  id: number; key: string; edition_id: number;
  game_id: string; game_label: string;            // "DET @ ORL"
  market: "spread"|"total"|"moneyline"|"player_pts"|"player_reb"|"player_ast"|"player_fg3m";
  market_label: string;                            // "ORL +2.5", "Over 213.5", "C. Cunningham Over 24.5 PTS"
  subject_type: "team"|"game"|"player"; subject_id: number|null; subject_name: string; team_abbr: string|null;
  side: "home"|"away"|"over"|"under"; line: number|null;
  price_american: number|null; book: string|null; quote_at: string|null;
  projection: number|null;                         // model point projection of the underlying stat
  prob: number|null; prob_low: number|null; prob_high: number|null;   // model probability + interval
  breakeven: number|null; edge: number|null;       // price-implied break-even prob; prob - breakeven
  sample_size: number|null;                        // historical observations behind the estimate
  evidence_score: number|null;
  evidence: { components: {name: string; score: number; max: number; note: string}[];
              supporting: string[]; opposing: string[] };
  classification: string; rejection_reasons: string[]; rank_in_game: number|null;
  model: { id: number; name: string; version: string; status: string } | null;
  verified_at: string; invalidated: boolean; invalidation_reason: string|null;
};

type ComboLeg = { selection_id: number; label: string; game_id: string; game_label: string; market: string;
  subject_name: string; side: string; line: number|null; price_american: number|null;
  prob: number|null; prob_low: number|null; prob_high: number|null; classification: string;
  evidence_score: number|null; sample_size: number|null; supporting: string[]; opposing: string[] };

type Combo = {
  uid: string; engine: "DOUBLE"|"TRIPLE"|"QUAD"|"FIVE"|"SIX"; n_legs: number;
  structure: "SGP"|"CROSS"|"MIXED"; name: string;                  // "GIZER TRIPLE #2"
  legs: ComboLeg[];
  joint_prob: number|null; joint_low: number|null; joint_high: number|null; independent_prob: number|null;
  correlation: { method: string; seed: number|null; draws: number|null; mc_se: number|null;
                 assessment: string; pairwise: {a: string; b: string; rho: number}[]; shared_risks: string[] };
  observed_quote: { decimal: number; american: number; book: string; at: string } | null;  // real quote only
  reference_price: { decimal: number; american: number; note: string } | null;  // arithmetic product of leg prices; NOT a quote
  evidence_score: number|null; weakest_assumption: string|null;
  classification: string; designation: "research_only"|"paper_testing";
  rejection_reasons: string[]; invalidation_conditions: string[];
  created_at: string; validated_at: string; invalidated: boolean; invalidation_reason: string|null;
};

type InjuryRow = { player_id: number|null; player_name: string; team_abbr: string|null; status: string;
  detail: string|null; comment: string|null; source: string; source_published_at: string|null; ingested_at: string };

type GameCard = {
  id: string; date: string; tipoff_utc: string|null; status: "scheduled"|"live"|"final"|"postponed";
  home: TeamRef; away: TeamRef; home_points: number|null; away_points: number|null; venue: string|null;
  label: string;
  injuries: { home: InjuryRow[]; away: InjuryRow[] };
  lineup_status: string;                             // e.g. "Starting lineups: no provider configured"
  research_completeness: { score: number; items: {name: string; ok: boolean; note: string}[] };
  environment: { proj_home_points: number; proj_away_points: number; proj_total: number; proj_margin: number;
                 proj_pace: number; total_p10: number; total_p90: number; notes: string[] } | null;
  quotes: { market: string; side: string; line: number|null; price_american: number|null; book: string; at: string }[];
  top_picks: Selection[];                            // 0..5 qualified selections, ranked
  n_evaluated: number; n_rejected: number; no_picks_reason: string|null;
};

type EditionRef = { id: number; key: string; kind: "morning"|"midday"|"afternoon"|"pregame"|"final";
  version: number; published_at: string };

type Ticket = { id: number; kind: "single"|"combo"; engine: string; structure: string; source: string;
  game_date: string; combo_uid: string|null; legs: ComboLeg[]; model_prob: number|null; prob_low: number|null;
  prob_high: number|null; price_decimal: number|null; price_basis: "observed_quote"|"leg_product_reference"|"none";
  classification: string; stake_units: number; created_at: string;
  settlement: { result: "win"|"loss"|"push"|"void"|"unresolved"; pnl_units: number|null; settled_at: string;
                leg_results: {label: string; result: string; actual: number|null}[];
                forensic: { verdict: string; findings: string[] } } | null };
```

### `/command-center`

```ts
{ date: string; is_today: boolean;
  available_dates: string[];                // dates with at least one edition or scheduled game (desc)
  edition: EditionRef | null; editions: EditionRef[];
  slate: { n_games: number; games: GameCard[] };
  counts: { markets_analyzed: number; qualified_selections: number; combinations: number;
            high_confidence: number; rejected: number };
  high_confidence: Selection[];             // may be empty; empty is a valid research outcome
  injury_changes: { text: string; at: string }[];
  latest_refresh_at: string | null;
  model_readiness: { family: string; name: string; version: string; status: string; ready: boolean; note: string }[];
  paper_performance: Summary;               // all-time
  agents: { ok: number; failed: number; never_run: number };
  findings: { finding_uid: string; hypothesis: string; status: string }[];
  warnings: string[] }
```

### `/results`

```ts
type Summary = { tickets: number; settled: number; wins: number; losses: number; pushes: number; voids: number;
  unresolved: number; hit_rate: number|null; hit_rate_ci: [number, number]|null; avg_model_prob: number|null;
  staked_units: number; pnl_units: number|null; roi: number|null; roi_ci: [number, number]|null;
  max_drawdown_units: number|null; priced_tickets: number };
{ summary: Summary;
  by_engine: ({ engine: string } & Summary)[];        // SINGLE, DOUBLE, TRIPLE, QUAD, FIVE, SIX — always all six
  by_structure: ({ structure: string } & Summary)[];  // SGP, CROSS
  calibration: { bucket: string; n: number; predicted: number; observed: number|null; ci: [number, number]|null }[];
  equity_curve: { date: string; pnl_units: number; cumulative: number }[];
  selection_outcomes: { n: number; note: string };
  items: Ticket[]; total: number;
  disclaimer: string }
```

### `/games/{id}` detail shapes

```ts
type TeamForm = { record: string; last_10: string; points_per_game: number; opp_points_per_game: number;
  off_rating: number|null; def_rating: number|null; pace: number|null; last_10_points_per_game: number;
  last_10_opp_points_per_game: number; games_in_sample: number } | {};          // {} when no history
type PlayerProj = { player_id: number; player_name: string; team_abbr: string|null; stat: "pts"|"reb"|"ast"|"fg3m";
  mean: number; p10: number; p50: number; p90: number; proj_min: number; n_games: number; injury_status: string|null };
```
`research_completeness.score` is 0–100. `/combos` `category` defaults to `double`. `POST /builder/evaluate`
accepts 1–6 selections.
